function getPin(){ return sessionStorage.getItem('relevoPin') || ''; }


async function loadData(){

  setStatus('Cargando…', '');

  try{

    const res = await fetch(APPS_SCRIPT_URL + '?action=read&pin=' + encodeURIComponent(getPin()));

    const json = await res.json();

    if(!json.ok){

      if(json.error === 'PIN incorrecto'){

        sessionStorage.removeItem('relevoPin');

        showLock('PIN incorrecto. Inténtalo de nuevo.');

        return false;

      }

      throw new Error(json.error || 'Error desconocido');

    }

    roomsData = json.data;

    if(json.capacities) capacities = Object.assign(capacities, json.capacities);

    render();

    setStatus('Actualizado', 'ok');

    document.getElementById('lastSync').textContent = 'Última actualización: ' + new Date().toLocaleTimeString('es-ES');

    return true;

  }catch(err){

    setStatus('Error al cargar: ' + err.message, 'err');

    return false;

  }

}

}

  if(!pin){ return; }

  sessionStorage.setItem('relevoPin', pin);

  document.getElementById('lockErr').textContent = '';

  const success = await loadData();

  if(success) hideLock();

}

function parseSvValue(raw){

  raw = (raw || '').toString().trim();

  if(/^control\s*de\s*diuresis$/i.test(raw) || /^control\s*de\s*diur[eé]sis$/i.test(raw)) return { fecha: '', tipo: '', cd: true };

  const m = raw.match(/^(.*?)\s*\(([^)]*)\)\s*$/);

  if(m) return { fecha: m[1].trim(), tipo: m[2].trim(), cd: false };

  return { fecha: raw, tipo: '', cd: false };

}


function composeSvValue(fecha, tipo){

  fecha = (fecha || '').trim();

  return !fecha ? '' : (tipo ? `${fecha} (${tipo})` : fecha);

}

    box.classList.add('dirty'); box.classList.remove('saved');

    scheduleSave(roomCode, 'svCheck', svCheckbox.checked, box);

  cdCheckbox.addEventListener('change', trigger);

}


function makeField(roomCode, field, value, htmlValue, isRich, grow, isInlineName){

  const box = document.createElement('div');

  box.className = 'field';

  const label = document.createElement('label');

  label.textContent = FIELD_LABELS[field];

  box.appendChild(label);


  if(RICHTEXT_FIELDS.includes(field)){

    const div = document.createElement('div');

    div.className = 'rich-field field-' + field + (isInlineName ? ' inline-name' : '');

    div.contentEditable = 'true';

    div.dataset.room = roomCode;

    div.dataset.field = field;

    if(htmlValue){ div.innerHTML = htmlValue; } else { div.textContent = value || ''; }


    div.addEventListener('input', () => {

      box.classList.add('dirty'); box.classList.remove('saved');

      scheduleSaveRich(div, box);

    });


    if(isInlineName || field === 'jc'){

      div.addEventListener('keydown', (e) => { if(e.key === 'Enter') e.preventDefault(); });

    }


    box.appendChild(div);

    attachCapacity(div, field, box);

    return box;

  }


  let input;

  if(field === 'especialidad'){

    input = makeEspecialidadSelect(roomCode, value);

    box.appendChild(input);

    return box;

  } else if(field === 'viaTipo'){

    input = document.createElement('select');

    input.className = 'via-select';

    VIA_TIPOS.forEach(opt => {

      const o = document.createElement('option');

      o.value = opt; o.textContent = opt || '—';

      if(String(value).toUpperCase() === opt.toUpperCase()) o.selected = true;

      input.appendChild(o);

    });

    applyViaColor(input);

  } else if(field === 'viaFecha'){

    input = document.createElement('input');

    input.type = 'text'; input.value = value || '';

    enforceDdMm(input);

  } else {

    input = document.createElement('input');

    input.type = 'text'; input.value = value || '';

  }


  input.addEventListener('input', () => {

    box.classList.add('dirty'); box.classList.remove('saved');

    scheduleSave(roomCode, field, input.value, box);

  });

  if(input.tagName === 'SELECT'){

    input.addEventListener('change', () => {

      box.classList.add('dirty'); box.classList.remove('saved');

      scheduleSave(roomCode, field, input.value, box);

      if(field === 'viaTipo') applyViaColor(input);

    });

  }


  box.appendChild(input);

  return box;

}

// ===== ESCALADO DE CAPACIDAD Y LÍMITE DE SALTOS DE LÍNEA =====

const SIZE_PX = { small: 11, normal: 13, large: 14 };


function sizeCost(sizeKey){

  const px = SIZE_PX[sizeKey] || 13;

  return (px / 13) * (px / 13);

}


function analyzeContent(div){

  const { runs, text } = extractRuns(div);

  let weighted = 0;

  let maxFontSize = 'normal';


  runs.forEach(r => { 

    weighted += (r.end - r.start) * sizeCost(r.size);

    if (r.size === 'large') maxFontSize = 'large';

    else if (r.size === 'normal' && maxFontSize !== 'large') maxFontSize = 'normal';

    else if (r.size === 'small' && maxFontSize === 'normal') maxFontSize = 'small';

  });


  const lineBreaks = (text.match(/\n/g) || []).length;


  return { weighted, lineBreaks, maxFontSize };

}


function attachCapacity(div, field, box){

  const limit = capacities[field] || 150;

  const counter = document.createElement('div');

  counter.className = 'char-counter';

  let lastValidHtml = div.innerHTML;


  function updateCounter(analysis){

    const maxBreaks = analysis.maxFontSize === 'small' ? 4 : 3;

    counter.textContent = Math.round(analysis.weighted) + ' / ' + limit + ' | Lín: ' + (analysis.lineBreaks + 1) + '/' + (maxBreaks + 1);

    counter.classList.toggle('over', analysis.weighted >= limit || analysis.lineBreaks > maxBreaks);

  }


  div.addEventListener('input', () => {

    const analysis = analyzeContent(div);

    const maxAllowedBreaks = analysis.maxFontSize === 'small' ? 4 : 3;


    if(analysis.weighted > limit || analysis.lineBreaks > maxAllowedBreaks){

      div.innerHTML = lastValidHtml;

      

      const range = document.createRange();

      range.selectNodeContents(div);

      range.collapse(false);

      const sel = window.getSelection();

      sel.removeAllRanges();

      sel.addRange(range);


      updateCounter(analyzeContent(div));

      return;

    }


    lastValidHtml = div.innerHTML;

    updateCounter(analysis);

  });


  updateCounter(analyzeContent(div));

  box.appendChild(counter);

}


function extractRuns(root){

  let text = '';

  const runs = [];


  function walk(node, bold, size){

    if(node.nodeType === Node.TEXT_NODE){

      const start = text.length;

      text += node.textContent;

      const end = text.length;

      if(end > start) runs.push({ start, end, bold, size });

      return;

    }

    if(node.nodeType !== Node.ELEMENT_NODE) return;


    const tag = node.tagName;

    if(tag === 'BR'){

      const start = text.length;

      text += '\n';

      runs.push({ start, end: text.length, bold, size });

      return;

    }


    let nb = bold, ns = size;

    if(tag === 'B' || tag === 'STRONG') nb = true;

    if(node.style && (node.style.fontWeight === 'bold' || node.style.fontWeight === '700')) nb = true;

    if(tag === 'FONT' && node.getAttribute('size')){

      const n = parseInt(node.getAttribute('size'), 10);

      if(n >= 4) ns = 'large';

      else if(n <= 2) ns = 'small';

      else ns = 'normal';

    }


    const isBlock = (tag === 'DIV' || tag === 'P') && node !== root;

    if(isBlock && text.length > 0 && !text.endsWith('\n')){

      const start = text.length;

      text += '\n';

      runs.push({ start, end: text.length, bold: false, size: 'normal' });

    }


    Array.from(node.childNodes).forEach(child => walk(child, nb, ns));

  }


  Array.from(root.childNodes).forEach(child => walk(child, false, 'normal'));


  const merged = [];

  runs.forEach(r => {

    const last = merged[merged.length - 1];

    if(last && last.bold === r.bold && last.size === r.size && last.end === r.start){

      last.end = r.end;

    } else {

      merged.push({ ...r });

    }

  });

  return { text, runs: merged };

}
